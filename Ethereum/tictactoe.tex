contract TicTacToe {
  uint constant public MINIMUM_BET = 0.01 ether;

  enum Tile { EMPTY, CROSS, CIRCLE }
  Tile[3][3] public board;

  address payable public owner;

  address payable public crossPlayer;

  address payable public circlePlayer;

  uint public bet;

  Tile public turn;

  constructor () {
    owner = payable(msg.sender);
    initGame();
  }

  function initGame() private {
    crossPlayer = payable(address(0));
    circlePlayer = payable(address(0));
    turn = Tile.CROSS;
    bet = 0;

    for (uint x = 1; x <= 3; x++)
      for (uint y = 1; y <= 3; y++)
        set(x, y, Tile.EMPTY);
  }

  function at(uint x, uint y) public view returns (Tile) {
    return board[x - 1][y - 1];
  }

  function set(uint x, uint y, Tile tile) private {
    board[x - 1][y - 1] = tile;
  }

  function play(uint x, uint y) payable public {
    require(1 <= x && x <= 3 && 1 <= y && y <= 3, "coordinates must be between 1 and 3");
    require(at(x, y) == Tile.EMPTY, "the selected tile is not empty");

    if (turn == Tile.CROSS)
      playCross(x, y);
    else
      playCircle(x, y);
  }

  function playCross(uint x, uint y) private {
    if (crossPlayer == address(0)) {
      require(msg.value >= MINIMUM_BET, "you must bet at least 0.01 ETH");
      crossPlayer = payable(msg.sender);
    }
    else
      require(msg.sender == crossPlayer, "it's not your turn");

    set(x, y, turn);
    bet += msg.value;

    if (isVictoryMove(x, y)) {
      crossPlayer.call{value: bet * 80 / 100}("");
      circlePlayer.call{value: bet * 5 / 100}("");
      owner.call{value: bet * 15 / 100}("");
      initGame();
    }
    else if (isFull()) {
      owner.call{value: bet}("");
      initGame();
    }
    else
      turn = Tile.CIRCLE;
  }

  function playCircle(uint x, uint y) private {
    if (circlePlayer == address(0)) {
      require(msg.sender != crossPlayer, "you cannot play against yourself");
      require(msg.value >= bet, "you must bet at least as much as the cross player");
      circlePlayer = payable(msg.sender);
    }
    else
      require(msg.sender == circlePlayer, "it's not your turn");

    set(x, y, turn);
    bet += msg.value;

    if (isVictoryMove(x, y)) {
      circlePlayer.call{value: bet * 80 / 100}("");
      crossPlayer.call{value: bet * 5 / 100}("");
      owner.call{value: bet * 15 / 100}("");
      initGame();
    }
    else if (isFull()) {
      owner.call{value: bet}("");
      initGame();
    }
    else
      turn = Tile.CROSS;
  }

  function isVictoryMove(uint x, uint y) private view returns (bool) {
    if (at(x, 1) == at(x, 2) && at(x, 2) == at(x, 3)) // column x
      return true;
    else if (at(1, y) == at(2, y) && at(2, y) == at(3, y)) // row y
      return true;
    else if (x == y && at(1, 1) == at(2, 2) && at(2, 2) == at(3, 3))
      return true;
    else if (x + y == 4 && at(1, 3) == at(2, 2) && at(2, 2) == at(3, 1))
      return true;
    else
      return false;
  }

  function isFull() private view returns (bool) {
    for (uint x = 1; x <= 3; x++)
      for (uint y = 1; y <= 3; y++)
        if (at(x, y) == Tile.EMPTY)
          return false;

    return true;
  }
}
